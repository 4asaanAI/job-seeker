# JobPilot M1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A multi-seeker job-discovery pipeline exposed as an MCP server, which finds brand-side merchandising roles, never reports a blocked source as "no jobs", and never notifies the same person about the same job twice.

**Architecture:** One Python MCP server (`jobpilot`) owning a SQLite store and all sources. Claude is the orchestrator and calls its tools; the server holds no model logic. Global `job_postings` are fetched once and shared; everything per-person hangs off `seeker_id`, enforced in the store layer rather than left to callers.

**Tech Stack:** Python 3.12, FastMCP, python-jobspy (MIT), SQLite (stdlib), PyYAML, Jinja2, pytest.

**Spec:** `docs/superpowers/specs/2026-09-19-jobpilot-design.md`

## Global Constraints

- **Python 3.12 only.** On the Windows desktop, pin `C:\Users\Desktop\AppData\Local\Programs\Python\Python312\python.exe`. Never rely on PATH: `where python` resolves to `C:\Python314` first.
- **Nothing installs to C:.** Deployment path is `E:\Github\Aasaan AI\jobpilot\`. Development happens on the Mac at `~/Desktop/Claude/jobpilot`.
- **Every per-seeker query filters on `seeker_id`**, enforced in the store layer. There is no "current seeker" global anywhere in the codebase.
- **`seeker_id` is never taken from message content.** It comes from a verified Telegram numeric sender ID or an explicit tool argument supplied by a scheduled run.
- **No network in tests.** Every source test runs against committed fixtures.
- **A source returning zero results is only "no jobs" if its canary passed.** Otherwise it is `BLOCKED` and must surface as such.
- Generated PowerShell scripts must be ASCII-only (PowerShell 5.1 reads UTF-8 as ANSI).
- Git operations on the desktop set `GCM_INTERACTIVE=Never`.

```python
# src/jobpilot/config.py
from __future__ import annotations

import tomllib
from dataclasses import dataclass
from pathlib import Path


@dataclass(frozen=True)
class CanaryConfig:
    site: str
    search_term: str
    location: str


@dataclass(frozen=True)
class SourceConfig:
    enabled: bool
    sites: list[str]
    country_indeed: str
    hours_old: int
    results_wanted: int
    canary: CanaryConfig


@dataclass(frozen=True)
class Config:
    data_dir: Path
    out_dir: Path
    score_batch: int
    sources: dict[str, SourceConfig]


def load_config(path: Path) -> Config:
    raw = tomllib.loads(Path(path).read_text())

    sources: dict[str, SourceConfig] = {}
    for name, block in raw.get("sources", {}).items():
        if "canary" not in block:
            raise ValueError(
                f"source '{name}' has no canary; a source without a canary "
                "cannot be distinguished from a blocked one"
            )
        sources[name] = SourceConfig(
            enabled=block["enabled"],
            sites=block["sites"],
            country_indeed=block["country_indeed"],
            hours_old=block["hours_old"],
            results_wanted=block["results_wanted"],
            canary=CanaryConfig(**block["canary"]),
        )

    return Config(
        data_dir=Path(raw["paths"]["data_dir"]),
        out_dir=Path(raw["paths"]["out_dir"]),
        score_batch=raw["limits"]["score_batch"],
        sources=sources,
    )
```

```python
# src/jobpilot/store/db.py
from __future__ import annotations

import hashlib
import re
import sqlite3
from datetime import datetime, timezone
from pathlib import Path

from .models import Posting, Seeker, SeekerJob

SCHEMA = Path(__file__).parent / "schema.sql"


def _now() -> str:
    return datetime.now(timezone.utc).isoformat()


def normalize_title(title: str) -> str:
    return re.sub(r"[^a-z0-9]+", " ", title.lower()).strip()


def content_hash(p: Posting) -> str:
    basis = f"{p.company_name}|{normalize_title(p.title)}|{p.location or ''}"
    return hashlib.sha256(basis.encode()).hexdigest()


class Store:
    def __init__(self, db_path: Path):
        self.db_path = Path(db_path)
        self.db_path.parent.mkdir(parents=True, exist_ok=True)
        self.conn = sqlite3.connect(self.db_path)
        self.conn.row_factory = sqlite3.Row
        self.conn.execute("PRAGMA foreign_keys = ON")

    def migrate(self) -> None:
        self.conn.executescript(SCHEMA.read_text())
        self.conn.commit()

    # --- seekers -------------------------------------------------------
    def add_seeker(
        self, name: str, telegram_user_id: int | None, profile_path: str, daily_cap: int
    ) -> int:
        cur = self.conn.execute(
            "INSERT INTO seekers (name, telegram_user_id, profile_path, daily_cap) "
            "VALUES (?, ?, ?, ?)",
            (name, telegram_user_id, profile_path, daily_cap),
        )
        self.conn.commit()
        return int(cur.lastrowid)

    def list_seekers(self) -> list[Seeker]:
        rows = self.conn.execute(
            "SELECT id, name, telegram_user_id, profile_path, active, daily_cap "
            "FROM seekers ORDER BY id"
        ).fetchall()
        return [
            Seeker(
                id=r["id"],
                name=r["name"],
                telegram_user_id=r["telegram_user_id"],
                profile_path=r["profile_path"],
                active=bool(r["active"]),
                daily_cap=r["daily_cap"],
            )
            for r in rows
        ]

    def seeker_by_telegram_id(self, telegram_user_id: int) -> Seeker | None:
        for s in self.list_seekers():
            if s.telegram_user_id == telegram_user_id:
                return s
        return None

    # --- postings ------------------------------------------------------
    def upsert_posting(self, p: Posting) -> int:
        now = _now()
        self.conn.execute(
            """
            INSERT INTO job_postings (
                company_name, source, external_id, title, normalized_title,
                location, description, url, posted_at,
                first_seen_at, last_seen_at, content_hash
            ) VALUES (?,?,?,?,?,?,?,?,?,?,?,?)
            ON CONFLICT(source, external_id) DO UPDATE SET
                last_seen_at = excluded.last_seen_at,
                title        = excluded.title,
                description  = excluded.description
            """,
            (
                p.company_name, p.source, p.external_id, p.title,
                normalize_title(p.title), p.location, p.description, p.url,
                p.posted_at, now, now, content_hash(p),
            ),
        )
        self.conn.commit()
        row = self.conn.execute(
            "SELECT id FROM job_postings WHERE source = ? AND external_id = ?",
            (p.source, p.external_id),
        ).fetchone()
        return int(row["id"])

    # --- seeker/posting links -----------------------------------------
    def link_posting(self, seeker_id: int, posting_id: int) -> None:
        self.conn.execute(
            "INSERT OR IGNORE INTO seeker_jobs (seeker_id, posting_id, first_seen_at) "
            "VALUES (?, ?, ?)",
            (seeker_id, posting_id, _now()),
        )
        self.conn.commit()

    def seeker_jobs(self, seeker_id: int, status: str | None = None) -> list[SeekerJob]:
        sql = (
            "SELECT sj.seeker_id, sj.posting_id, sj.status, "
            "       jp.title, jp.company_name, jp.url "
            "FROM seeker_jobs sj JOIN job_postings jp ON jp.id = sj.posting_id "
            "WHERE sj.seeker_id = ?"
        )
        params: list[object] = [seeker_id]
        if status is not None:
            sql += " AND sj.status = ?"
            params.append(status)
        sql += " ORDER BY sj.posting_id"
        return [
            SeekerJob(
                seeker_id=r["seeker_id"],
                posting_id=r["posting_id"],
                status=r["status"],
                title=r["title"],
                company_name=r["company_name"],
                url=r["url"],
            )
            for r in self.conn.execute(sql, params).fetchall()
        ]
```

```python
    # --- scores --------------------------------------------------------
    def record_score(
        self, seeker_id: int, posting_id: int, fit_score: float,
        reasoning: str, model: str | None = None,
    ) -> None:
        self.conn.execute(
            """
            INSERT INTO scores (seeker_id, posting_id, fit_score, reasoning, model, scored_at)
            VALUES (?,?,?,?,?,?)
            ON CONFLICT(seeker_id, posting_id) DO UPDATE SET
                fit_score = excluded.fit_score,
                reasoning = excluded.reasoning,
                model     = excluded.model,
                scored_at = excluded.scored_at
            """,
            (seeker_id, posting_id, fit_score, reasoning, model, _now()),
        )
        self.conn.execute(
            "UPDATE seeker_jobs SET status = 'scored' "
            "WHERE seeker_id = ? AND posting_id = ? AND status = 'discovered'",
            (seeker_id, posting_id),
        )
        self.conn.commit()

    def unscored_jobs(self, seeker_id: int, limit: int) -> list[SeekerJob]:
        rows = self.conn.execute(
            """
            SELECT sj.seeker_id, sj.posting_id, sj.status, jp.title,
                   jp.company_name, jp.url
            FROM seeker_jobs sj
            JOIN job_postings jp ON jp.id = sj.posting_id
            LEFT JOIN scores s
                   ON s.seeker_id = sj.seeker_id AND s.posting_id = sj.posting_id
            WHERE sj.seeker_id = ? AND s.posting_id IS NULL
            ORDER BY sj.posting_id
            LIMIT ?
            """,
            (seeker_id, limit),
        ).fetchall()
        return [

            JOIN job_postings jp ON jp.id = sj.posting_id
```

```python
# src/jobpilot/sources/jobspy_source.py
from __future__ import annotations

from typing import Any, Callable

from ..config import SourceConfig
from ..store.models import Posting


def _default_scraper(**kwargs: Any) -> list[dict]:
    from jobspy import scrape_jobs

    df = scrape_jobs(**kwargs)
    if df is None or len(df) == 0:
        return []
    return df.fillna("").to_dict("records")


class JobSpySource:
    name = "jobspy"

    def __init__(self, cfg: SourceConfig, scraper: Callable[..., list[dict]] | None = None):
        self.cfg = cfg
        self._scraper = scraper or _default_scraper

    def _scrape(self, sites: list[str], term: str, location: str, wanted: int) -> list[dict]:
        return self._scraper(
            site_name=sites,
            search_term=term,
            location=location,
            results_wanted=wanted,
            hours_old=self.cfg.hours_old,
            country_indeed=self.cfg.country_indeed,
        )

    def search(self, term: str, location: str) -> list[Posting]:
        rows = self._scrape(self.cfg.sites, term, location, self.cfg.results_wanted)
        postings: list[Posting] = []
        for row in rows:
            external_id = str(row.get("id") or "").strip()
            title = str(row.get("title") or "").strip()
            company = str(row.get("company") or "").strip()
            if not external_id or not title or not company:
                continue
            postings.append(
                Posting(
                    company_name=company,
                    source=str(row.get("site") or self.name),
                    external_id=external_id,
                    title=title,
                    location=str(row.get("location") or "") or None,
                    description=str(row.get("description") or "") or None,
                    url=str(row.get("job_url") or "") or None,
                    posted_at=str(row.get("date_posted") or "") or None,
                )
            )
        return postings

    def canary(self) -> bool:
        c = self.cfg.canary
        try:
            rows = self._scrape([c.site], c.search_term, c.location, 5)
        except Exception:
            return False
        return len(rows) > 0
```

```python
# src/jobpilot/sources/health.py
from __future__ import annotations

import statistics

from .base import SourceStatus

DEGRADED_FRACTION = 0.25
MIN_HISTORY = 3


def classify(
    results_count: int, canary_ok: bool, history: list[int]
) -> tuple[SourceStatus, str]:
    """Decide what a result count actually means.

    The rule that matters: a zero result is only 'no jobs' when the canary
    passed. Otherwise the source is blocked and its silence is meaningless.
    """
    if not canary_ok:
        return (
            SourceStatus.BLOCKED,
            "Canary returned nothing: source is blocked or rate-limited. "
            "Its zero/low result count carries no information.",
        )

    usable = [h for h in history if h > 0]
    if len(usable) >= MIN_HISTORY:
        median = statistics.median(usable)
        if median > 0 and results_count < median * DEGRADED_FRACTION:
            return (
                SourceStatus.DEGRADED,
                f"Returned {results_count} against a recent median of {median:.0f}.",
            )

    return (SourceStatus.HEALTHY, f"Returned {results_count} results.")
```

```python
        sql = (

            "FROM source_runs WHERE id IN "

        )
        params: list[object] = []
        if since is not None:
            sql += " AND run_at >= ?"
            params.append(since)
        return [dict(r) for r in self.conn.execute(sql, params).fetchall()]
```

```python
# src/jobpilot/profiles.py
from __future__ import annotations

import csv
from dataclasses import dataclass
from pathlib import Path

import yaml

REQUIRED = ("name", "headline", "location", "search_terms")


@dataclass(frozen=True)
class Profile:
    name: str
    headline: str
    location: str
    accounts: list[str]
    search_terms: list[str]
    target_companies: list[str]
    min_score: float


@dataclass(frozen=True)
class CompanyRow:
    name: str
    tier: int
    ats_vendor: str
    careers_url: str
    region: str


def load_profile(path: Path) -> Profile:
    raw = yaml.safe_load(Path(path).read_text()) or {}
    missing = [f for f in REQUIRED if not raw.get(f)]
    if missing:
        raise ValueError(f"profile {path} missing required fields: {', '.join(missing)}")
    return Profile(
        name=raw["name"],
        headline=raw["headline"],
        location=raw["location"],
        accounts=list(raw.get("accounts", [])),
        search_terms=list(raw["search_terms"]),
        target_companies=list(raw.get("target_companies", [])),
        min_score=float(raw.get("min_score", 70)),
    )


def load_companies(path: Path) -> list[CompanyRow]:
    rows: list[CompanyRow] = []
    with Path(path).open(newline="", encoding="utf-8") as fh:
        for r in csv.DictReader(fh):
            if r.get("enabled", "1").strip() != "1":
                continue
            rows.append(
                CompanyRow(
                    name=r["name"].strip(),
                    tier=int(r["tier"]),
                    ats_vendor=(r.get("ats_vendor") or "").strip(),
                    careers_url=(r.get("careers_url") or "").strip(),
                    region=(r.get("region") or "").strip(),
                )
            )
    return rows
```

```python
# src/jobpilot/pipeline.py
from __future__ import annotations

from .profiles import Profile
from .sources.base import SourceResult, SourceStatus
from .sources.health import classify
from .sources.jobspy_source import JobSpySource
from .store.db import Store


def run_discovery(
    store: Store, source: JobSpySource, seeker_id: int, profile: Profile
) -> SourceResult:
    """Fetch, classify, and store. Never let a blocked source look empty."""
    canary_ok = source.canary()

    postings = []
    if canary_ok:
        seen: set[tuple[str, str]] = set()
        for term in profile.search_terms:
            try:
                for p in source.search(term, profile.location):
                    key = (p.source, p.external_id)
                    if key not in seen:
                        seen.add(key)
                        postings.append(p)
            except Exception:
                canary_ok = False
                postings = []
                break

    history = store.recent_counts(source.name, 7)
    status, notes = classify(len(postings), canary_ok, history)

    if status is not SourceStatus.BLOCKED:
        for p in postings:
            posting_id = store.upsert_posting(p)
            store.link_posting(seeker_id, posting_id)

    store.record_source_run(
        source.name, canary_ok, len(postings), status.value, notes
    )
    return SourceResult(
        source=source.name, status=status, canary_ok=canary_ok,
        postings=postings, notes=notes,
    )
```

```python
# src/jobpilot/digest/render.py
from __future__ import annotations

from datetime import datetime, timezone
from pathlib import Path

from jinja2 import Environment, select_autoescape

from ..store.db import Store

TEMPLATE = """<!doctype html>
<html><head><meta charset="utf-8"><title>JobPilot digest - {{ seeker.name }}</title>
<style>
 body{font-family:system-ui,sans-serif;max-width:46rem;margin:2rem auto;padding:0 1rem}
 .job{border:1px solid #ddd;border-radius:8px;padding:1rem;margin:1rem 0}
 .score{font-weight:700}
 .health{background:#f6f6f6;border-radius:8px;padding:.75rem;font-size:.9rem}
 .blocked{background:#ffe9e9}
</style></head><body>
<h1>{{ seeker.name }}</h1>
<p>{{ generated_at }}</p>

<div class="health {% if blocked %}blocked{% endif %}">
{% if blocked %}
  <strong>Some sources could not be read this run.</strong>
  Results below are incomplete - this is not a quiet market.
  <ul>{% for h in health if h.status != 'healthy' %}
    <li>{{ h.source }}: {{ h.status }} - {{ h.notes }}</li>
  {% endfor %}</ul>
{% else %}
  All sources healthy.
  {% for h in health %}{{ h.source }}: {{ h.results_count }} results. {% endfor %}
{% endif %}
</div>

{% if jobs %}
  <h2>{{ jobs|length }} role{{ '' if jobs|length == 1 else 's' }} worth a look</h2>
  {% for j in jobs %}
  <div class="job">
    <div class="score">{{ '%.0f'|format(j.fit_score) }}/100</div>
    <h3><a href="{{ j.url }}">{{ j.title }}</a></h3>
    <p>{{ j.company_name }}</p>
    <p>{{ j.reasoning }}</p>
  </div>
  {% endfor %}
{% elif not blocked %}
  <p>Nothing cleared the bar this run. Sources were healthy, so this is a
  genuinely quiet period rather than a gap in coverage.</p>
{% endif %}
</body></html>
"""


def render_digest(
    store: Store, seeker_id: int, min_score: float, limit: int, out_dir: Path
) -> Path:
    seeker = next(s for s in store.list_seekers() if s.id == seeker_id)
    health = store.source_health_summary()
    blocked = any(h["status"] in ("blocked", "degraded", "error") for h in health)

    rows = store.conn.execute(
        """
        SELECT jp.title, jp.company_name, jp.url, s.fit_score, s.reasoning
        FROM seeker_jobs sj
        JOIN job_postings jp ON jp.id = sj.posting_id
        JOIN scores s ON s.seeker_id = sj.seeker_id AND s.posting_id = sj.posting_id
        WHERE sj.seeker_id = ? AND s.fit_score >= ?
        ORDER BY s.fit_score DESC LIMIT ?
        """,
        (seeker_id, min_score, limit),
    ).fetchall()

    env = Environment(autoescape=select_autoescape(default_for_string=True))
    html = env.from_string(TEMPLATE).render(
        seeker=seeker,
        jobs=[dict(r) for r in rows],
        health=health,
        blocked=blocked,
        generated_at=datetime.now(timezone.utc).strftime("%Y-%m-%d %H:%M UTC"),
    )

    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    out = out_dir / f"digest-seeker-{seeker_id}.html"
    out.write_text(html, encoding="utf-8")
    return out
```

```python
# src/jobpilot/server.py
from __future__ import annotations

import os
from pathlib import Path

from .config import load_config
from .digest.render import render_digest
from .pipeline import run_discovery
from .profiles import load_profile
from .sources.jobspy_source import JobSpySource
from .store.db import Store


def build_tools(store: Store, out_dir: Path) -> dict:
    """Plain callables, so they are testable without an MCP client."""

    def list_seekers() -> list[dict]:
        return [
            {"seeker_id": s.id, "name": s.name, "active": s.active}
            for s in store.list_seekers()
        ]

    def resolve_telegram_sender(telegram_user_id: int) -> dict:
        seeker = store.seeker_by_telegram_id(int(telegram_user_id))
        if seeker is None:
            return {
                "seeker_id": None,
                "error": "Sender not recognised. Refuse the request.",
            }
        return {"seeker_id": seeker.id, "name": seeker.name}

    def discover(seeker_id: int, config_path: str = "config.toml") -> dict:
        cfg = load_config(Path(config_path))
        seeker = next(s for s in store.list_seekers() if s.id == seeker_id)
        profile = load_profile(Path(seeker.profile_path))
        source = JobSpySource(cfg.sources["jobspy"])
        result = run_discovery(store, source, seeker_id, profile)
        return {
            "status": result.status.value,
            "found": len(result.postings),
            "notes": result.notes,
        }

    def get_unscored(seeker_id: int, limit: int = 40) -> list[dict]:
        return [
            {
                "posting_id": j.posting_id,
                "title": j.title,
                "company": j.company_name,
                "url": j.url,
            }
            for j in store.unscored_jobs(seeker_id, limit)
        ]

    def record_score(
        seeker_id: int, posting_id: int, fit_score: float, reasoning: str
    ) -> dict:
        store.record_score(seeker_id, posting_id, fit_score, reasoning, "claude")
        return {"ok": True}

    def get_notifiable(seeker_id: int, min_score: float = 70.0, limit: int = 5) -> list[dict]:
        return [
            {
                "posting_id": j.posting_id,
                "title": j.title,
                "company": j.company_name,
                "url": j.url,
            }
            for j in store.notifiable_postings(seeker_id, min_score, limit)
        ]

    def record_notification(
        seeker_id: int, posting_id: int, channel: str, kind: str
    ) -> dict:
        store.record_notification(seeker_id, posting_id, channel, kind)
        return {"ok": True}

    def source_health() -> list[dict]:
        return store.source_health_summary()

    def build_digest(seeker_id: int, min_score: float = 70.0, limit: int = 5) -> dict:
        path = render_digest(store, seeker_id, min_score, limit, out_dir)
        return {"path": str(path)}

    return {
        "list_seekers": list_seekers,
        "resolve_telegram_sender": resolve_telegram_sender,
        "discover": discover,
        "get_unscored": get_unscored,
        "record_score": record_score,
        "get_notifiable": get_notifiable,
        "record_notification": record_notification,
        "source_health": source_health,
        "build_digest": build_digest,
    }


def main() -> None:
    from fastmcp import FastMCP

    db_path = Path(os.environ.get("JOBPILOT_DB", "data/jobpilot.db"))
    out_dir = Path(os.environ.get("JOBPILOT_OUT", "out"))

    store = Store(db_path)
    store.migrate()

    mcp = FastMCP("jobpilot")
    for name, fn in build_tools(store, out_dir).items():
        mcp.tool(name=name)(fn)
    mcp.run()


if __name__ == "__main__":
    main()
```
